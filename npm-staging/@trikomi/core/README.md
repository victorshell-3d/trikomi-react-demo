# @trikomi/core

Unified 3D Engine and Addons for Trikomi 3D Platform.

## Usage

```javascript
// Core Engine
import { ThreeViewer, CoreState } from '@trikomi/core';

// Application Addons (subpath imports)
import { EyewearPlugin } from '@trikomi/core/eyewear';
import { BoxConfigurator } from '@trikomi/core/box';
import { SportswearCompositor } from '@trikomi/core/sportswear';
import { EighthWallAR } from '@trikomi/core/8thwall';
import { FaceMocap } from '@trikomi/core/face-mocap';
import { Tour360 } from '@trikomi/core/tour';
import { ViewerPlugin } from '@trikomi/core/viewer';
```
